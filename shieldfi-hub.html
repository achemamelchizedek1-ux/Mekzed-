Quick Start
# Start the backend
node backend/src/standalone-server.js

# Open the platform hub
open frontend/index.html

# Or launch unified multi-role app
open frontend/index-app.html

# Test the core ZK verify endpoint
curl -X POST http://localhost:3001/api/zk/verify \
  -H "Content-Type: application/json" \
  -d '{"proof":{"pi_a":["1","2","1"],"pi_b":[["3","4"],["5","6"],["1","0"]],"pi_c":["7","8","1"],"protocol":"groth16","curve":"bn128"},"publicSignals":["1000","100000","0xabc"]}'
